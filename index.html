☁ Supabase (sincronização entre dispositivos)

Rode o SQL abaixo no SQL Editor do Supabase, depois cole a Project URL e a chave anon public em cada dispositivo.

create table pedidos_antecipados_db (
  id text primary key default 'main',
  data jsonb not null default '[]'::jsonb,
  updated_at timestamptz default now()
);
alter table pedidos_antecipados_db enable row level security;
create policy "allow anon all" on pedidos_antecipados_db
  for all to anon using (true) with check (true);
